<script>
        // Progress Logic
        const circle = document.getElementById('progress-circle');
        const text = document.getElementById('percentage-text');
        const status = document.getElementById('loading-status');
        const footer = document.getElementById('footer-quote');
        const radius = circle.r.baseVal.value;
        const circumference = radius * 2 * Math.PI;

        circle.style.strokeDasharray = `${circumference} ${circumference}`;
        circle.style.strokeDashoffset = circumference;

        function setProgress(percent) {
            const offset = circumference - (percent / 100 * circumference);
            circle.style.strokeDashoffset = offset;
            text.innerText = `${Math.round(percent)}%`;
        }

        const statuses = [
            "Initializing precision engine",
            "Warming up mechanical switches",
            "Synthesizing focus layers",
            "Synchronizing leaderboard data",
            "Ready for high-performance"
        ];

        let currentPercent = 0;
        const duration = 4000; // 4 seconds loading
        const startTime = Date.now();

        function update() {
            const elapsed = Date.now() - startTime;
            currentPercent = Math.min((elapsed / duration) * 100, 100);
            
            setProgress(currentPercent);

            // Update status text based on progress
            const statusIndex = Math.floor((currentPercent / 101) * statuses.length);
            if (statuses[statusIndex]) {
                status.innerText = statuses[statusIndex];
            }

            if (currentPercent < 100) {
                requestAnimationFrame(update);
            } else {
                status.innerText = "Access Granted";
                status.classList.add('text-tertiary');
                circle.classList.replace('text-primary', 'text-tertiary');
                
                // Final interaction hint
                setTimeout(() => {
                    document.body.style.cursor = 'pointer';
                }, 500);
            }
        }

        // Show footer after delay
        setTimeout(() => {
            footer.classList.remove('opacity-0');
            footer.classList.add('opacity-100');
        }, 1200);

        // Start animation
        requestAnimationFrame(update);

        // Micro-interaction: Mouse movement subtle parallax on atmospheric light
        document.addEventListener('mousemove', (e) => {
            const x = (e.clientX / window.innerWidth - 0.5) * 20;
            const y = (e.clientY / window.innerHeight - 0.5) * 20;
            document.querySelectorAll('.blur-full').forEach(el => {
                el.style.transform = `translate(${x}px, ${y}px)`;
            });
        });
    </script>
