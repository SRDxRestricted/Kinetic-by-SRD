        // Micro-interaction for Key Press Confirmation
        document.addEventListener('keydown', (e) => {
            // Visual feedback on stage
            const stage = document.getElementById('typing-stage');
            stage.style.transform = 'scale(0.998)';
            setTimeout(() => stage.style.transform = 'scale(1)', 50);

            // Mock update stats for visual flair
            if (Math.random() > 0.8) {
                const wpm = document.getElementById('stat-wpm');
                const current = parseInt(wpm.innerText);
                wpm.innerText = current + (Math.random() > 0.5 ? 1 : -1);
            }
        });

        // Always keep input focused
        document.addEventListener('click', () => {
            document.getElementById('typing-input').focus();
        });

        // Progress bar simulation
        let progress = 35;
        setInterval(() => {
            progress = (progress + 0.1) % 100;
            document.getElementById('test-progress').style.width = progress + '%';
        }, 1000);
